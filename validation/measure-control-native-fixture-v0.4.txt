  window.__kwMeasureBoundary=(caseId)=>{
    const r={caseId};
    const snapshot=()=>({
      round:state.combat.round,records:meleeGroups().map(g=>({
        id:g.meleeId,range:g.range,formedRound:g.formedRound,
        pcNames:g.pcNames.slice(),monsterIds:g.monsterIds.slice(),
        sides:meleeParticipatingSideIds(g),
        sideIntents:Object.fromEntries(meleeParticipatingSideIds(g).map(s=>[s,meleeSideIntent(g,s)]))
      }))
    });
    const setup=(count=1,npcs=1)=>{
      installHarnessFixture({distance:0,count});
      state.party=state.party.slice(0,npcs);
      state.combat.monsters=state.combat.monsters.slice(0,count);
      state.rules.initiativeMode='PHASED';
      for(let i=0;i<state.party.length;i++)
        setCombatPcVector(state.party[i],{x:100*i,y:0,z:0});
      for(let i=0;i<state.combat.monsters.length;i++)
        setCombatMonsterVector(state.combat.monsters[i],{x:4+100*i,y:0,z:0});
      return {pc:state.party[0],foe:state.combat.monsters[0]};
    };
    if(caseId==='first_contact'){
      const {pc,foe}=setup();
      r.initial=setEngagement(pc,foe,'far');
      state.combat.initiative={rolls:{pc:4,foes:5},pc:4,monster:5};
      state.combat.pcOrders={[pc.name.toLowerCase()]:{rangeIntent:'hold',movement:'hold',action:'parry'}};
      state.combat.monsterOrders={[foe.id]:{rangeIntent:'press',targetName:pc.name,action:'melee'}};
      r.before=snapshot();
      resolveMeleeDistanceContests();r.after=snapshot();
      return r;
    }
    if(caseId==='far_hold_legacy'||caseId==='normal_hold_legacy'){
      const {pc,foe}=setup();

      state.combat.initiative={rolls:{pc:4,foes:5},pc:4,monster:5};
      state.combat.pcOrders={[pc.name.toLowerCase()]:{rangeIntent:'hold',movement:'hold',action:'parry'}};
      state.combat.monsterOrders={[foe.id]:{rangeIntent:'press',targetName:pc.name,action:'melee'}};
      r.before=snapshot();resolveMeleeDistanceContests();r.after=snapshot();
      r.distanceResolveMarker=state.combat.meleeDistanceResolvedRound;
      resolveMeleeDistanceContests();r.afterDuplicate=snapshot();
      return r;
    }
    if(caseId==='positions_and_barriers'){
      const {pc,foe}=setup();
      r.contact=atMeleeContact(pc,foe);
      r.firstContact=setEngagement(pc,foe,'far');
      r.groupsAtContact=snapshot().records.length;
      setCombatMonsterVector(foe,{x:11,y:0,z:0});
      r.contactAfterMove=atMeleeContact(pc,foe);
      r.groupsAfterMove=snapshot().records.length;
      setCombatMonsterVector(foe,{x:4,y:0,z:0});
      state.combat.spatialBarriers=[{minX:1,maxX:3,minY:-2,maxY:2,minZ:-2,maxZ:2,blocksEngagement:true}];
      r.contactBlocked=atMeleeContact(pc,foe);
      r.groupsBlocked=snapshot().records.length;
      state.combat.spatialBarriers=[];
      r.contactRestored=atMeleeContact(pc,foe);
      r.groupsRestored=snapshot().records.length;
      return r;
    }
    if(caseId==='two_independent_melees'){
      const {pc,foe}=setup(2,2);
      setEngagement(pc,foe,'far');
      const second=state.party[1],other=state.combat.monsters[1];
      setEngagement(second,other,'close');
      r.snapshot=snapshot();
      r.sideRecords=combatSideRecords().map(s=>s.id);
      return r;
    }
    if(caseId==='connected_multi_actor'){
      const {pc,foe}=setup(1,2);
      setCombatPcVector(state.party[1],{x:0,y:0,z:0});
      setEngagement(pc,foe,'far');
      setEngagement(state.party[1],foe,'far');
      state.combat.pcOrders={
        [pc.name.toLowerCase()]:{rangeIntent:'hold',action:'parry',movement:'hold'},
        [state.party[1].name.toLowerCase()]:{rangeIntent:'press',action:'melee',movement:'hold'}
      };
      state.combat.monsterOrders={[foe.id]:{rangeIntent:'press',action:'melee',targetName:pc.name}};
      r.snapshot=snapshot();
      return r;
    }
    if(caseId==='real_retreat'){
      const {pc,foe}=setup();
      setEngagement(pc,foe,'close');
      r.before=snapshot();
      r.start=combatPcVector(pc);
      state.combat.pcOrders={[pc.name.toLowerCase()]:{rangeIntent:'hold',movement:'retreat',action:'hold'}};

      r.after=snapshot();r.separation=combatSeparation(pc,foe);

    }
    if(caseId==='initiative_modifiers'){
      const {pc,foe}=setup();
      setEngagement(pc,foe,'far');
      state.combat.initiative={rolls:{pc:7,foes:0},pc:7,monster:0};

    }
    if(caseId==='phased_handoff'||caseId==='raw_handoff'){
      const {pc,foe}=setup();
      setEngagement(pc,foe,'far');
      state.combat.round=2;
      state.combat.initiative={
        rolls:{pc:4,foes:5},pc:4,monster:5,order:['foes','pc'],
        winner:'foes',winners:['foes']

      state.combat.pcOrders={[pc.name.toLowerCase()]:{
        rangeIntent:'hold',movement:'hold',action:'hold'
      }};

        rangeIntent:'press',movement:'hold',action:'hold'
      }};

      if(r.mode==='PHASED'){
        state.combat.moveOrder=['foes','pc'];
        resolvePhasedRoundAfterMovementChoice();
      }else{

      }

        round:state.combat.round,

        range:snapshot().records?.[0]?.range,
        awaitingWeaponChanges:state.combat.awaitingWeaponChanges,
        pending:state.combat.pendingResolution?.mode??null
      };
      if(state.combat.awaitingWeaponChanges)continueCombatAfterWeaponChanges(true);
      r.after=snapshot();r.afterAwaiting=state.combat.awaitingWeaponChanges;
      return r;
    }
    throw new Error('Unknown native test '+caseId);
  };
